### 参数说明
| 参数名      | 说明                    |
| -------- | --------------------- |
| phone    | 必要参数：用户手机号，获取短息验证码。   |
| captcha  | 必要参数：短信验证码。           |
| password | 必要参数：使用Crypt加密。       |
| roleId   | 必要参数：用户类型（0：学生，1：老师）。 |
| serverId | 必要参数：平台ID。            |
| province | 可选参数：省id。             |
| city     | 可选参数：市id。             |
| area     | 可选参数：区/县id            |

#### 密码加密方式
```html
import CryptoJS from 'crypto-js'
// 加密
const keyStr = '73H6G9X8HN1NY810' // 解密用的key
export function encrypt (word) {
  const time = Date.now()

  const key = CryptoJS.enc.Utf8.parse(keyStr)
  const srcs = CryptoJS.enc.Utf8.parse(time + word) // 加密方式: 时间戳 + 密文
  const encrypted = CryptoJS.AES.encrypt(srcs, key, {
    mode: CryptoJS.mode.ECB,
    padding: CryptoJS.pad.Pkcs7
  })
  return encrypted.toString()
}
```

### 后端拦截器
``` java
    /**
     * 根据accessToken 获取用户信息
     * @param accessToken accessToken
     * @param needDecrypt 是否需要解密
     * @return 用户信息
     */
    public UserInfoInTokenBO getUserInfoByAccessToken(String accessToken, boolean needDecrypt) {
        if (StrUtil.isBlank(accessToken)) {
            throw new Exception(ResponseEnum.UNAUTHORIZED,"accessToken is blank");
        }
        String keyName = OauthCacheNames.USER_INFO + accessToken;
        Object redisCache = redisTemplate.opsForValue().get(keyName);
        if (redisCache == null) {
            throw new Exception(ResponseEnum.UNAUTHORIZED,"登录过期，请重新登录");
        }
        return JSON.parseObject(redisCache.toString(), UserInfoInTokenBO.class);
    }
```
